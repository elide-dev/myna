cmake_minimum_required(VERSION 3.20)
project(myna_llvm LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 23)
  message(FATAL_ERROR "myna-llvm requires the same LLVM 23 toolchain as the participating compilers")
endif()
add_executable(myna-llvm main.cpp)
target_compile_features(myna-llvm PRIVATE cxx_std_17)
target_include_directories(myna-llvm SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_options(myna-llvm PRIVATE ${LLVM_DEFINITIONS_LIST})
if(TARGET LLVM)
  target_link_libraries(myna-llvm PRIVATE LLVM)
else()
  llvm_map_components_to_libnames(LLVM_LIBS Core IRReader BitReader BitWriter Analysis Object Support TransformUtils)
  target_link_libraries(myna-llvm PRIVATE ${LLVM_LIBS})
endif()
install(TARGETS myna-llvm RUNTIME DESTINATION bin)
