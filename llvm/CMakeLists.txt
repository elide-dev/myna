cmake_minimum_required(VERSION 3.20)
project(svmgen_llvm LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 23)
  message(FATAL_ERROR "svmgen-llvm requires the same LLVM 23 toolchain as the participating compilers")
endif()
add_executable(svmgen-llvm main.cpp)
target_compile_features(svmgen-llvm PRIVATE cxx_std_17)
target_include_directories(svmgen-llvm SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_options(svmgen-llvm PRIVATE ${LLVM_DEFINITIONS_LIST})
if(TARGET LLVM)
  target_link_libraries(svmgen-llvm PRIVATE LLVM)
else()
  llvm_map_components_to_libnames(LLVM_LIBS Core IRReader BitReader BitWriter Analysis Object Support)
  target_link_libraries(svmgen-llvm PRIVATE ${LLVM_LIBS})
endif()
install(TARGETS svmgen-llvm RUNTIME DESTINATION bin)
