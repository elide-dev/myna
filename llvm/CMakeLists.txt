cmake_minimum_required(VERSION 3.20)
project(myna_llvm LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 23)
  message(FATAL_ERROR "myna-llvm requires the same LLVM 23 toolchain as the participating compilers")
endif()
add_executable(myna-llvm main.cpp)
target_compile_features(myna-llvm PRIVATE cxx_std_17)
target_compile_options(myna-llvm PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(myna-llvm PRIVATE -Wl,--build-id=sha1)
endif()
# Mach-O UUIDs do not cover DWARF; keep one UUID from naming two different dSYMs.
include("${CMAKE_CURRENT_LIST_DIR}/debug-identity.cmake")
myna_bind_debug_identity(myna-llvm)
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(myna-llvm PRIVATE -fno-rtti)
endif()
target_include_directories(myna-llvm SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_options(myna-llvm PRIVATE ${LLVM_DEFINITIONS_LIST})
if(TARGET LLVM)
  target_link_libraries(myna-llvm PRIVATE LLVMObjCopy LLVM)
else()
  llvm_map_components_to_libnames(LLVM_LIBS Core IRReader BitReader BitWriter Analysis Object ObjCopy Support TransformUtils)
  target_link_libraries(myna-llvm PRIVATE ${LLVM_LIBS})
endif()
install(TARGETS myna-llvm RUNTIME DESTINATION bin)

option(MYNA_COVERAGE "Instrument the applicator for native coverage" OFF)
if(MYNA_COVERAGE)
  target_compile_options(myna-llvm PRIVATE -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(myna-llvm PRIVATE -fprofile-instr-generate)
  add_executable(myna-llvm-unit main.cpp)
  target_compile_definitions(myna-llvm-unit PRIVATE MYNA_LLVM_UNIT_TEST=1)
  target_compile_features(myna-llvm-unit PRIVATE cxx_std_17)
  target_include_directories(myna-llvm-unit SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
  get_target_property(MYNA_OPTIONS myna-llvm COMPILE_OPTIONS)
  target_compile_options(myna-llvm-unit PRIVATE ${MYNA_OPTIONS})
  get_target_property(MYNA_LIBRARIES myna-llvm LINK_LIBRARIES)
  target_link_libraries(myna-llvm-unit PRIVATE ${MYNA_LIBRARIES})
  target_link_options(myna-llvm-unit PRIVATE -fprofile-instr-generate)
endif()
